# 1. Ensure Node.js is installed (preferably v18+)
node -v
npm -v

# 2. Create a Vite project with React + TypeScript
npm create vite@latest my-app -- --template react-ts

# 3. Move into the project folder
cd my-app

# 4. Install nodemodules
npm install

# 5. Run the development server
npm run dev

# 6.  Initialize Git
git init
